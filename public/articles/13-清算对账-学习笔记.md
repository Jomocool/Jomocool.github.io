## 三、核心设计

### 1. 内部对账（同系统内核对）

用「成交明细」重算「持仓」（买入 − 卖出），与记录持仓对比。不一致说明数据漂移。

### 2. 外部对账（与交易所核对）

两份成交列表按 `tradeId` 关联，找出三种差异：

| 差异 | 含义 |
|------|------|
| 缺失 missing | 交易所有、我方无（漏记） |
| 多余 extra | 我方有、交易所无（多记） |
| 金额不符 mismatch | 都有但价格/数量对不上 |

### 3. K 线聚合（OHLCV）

把成交按**时间桶**（如 1 分钟）聚合：
`开`=第一笔价、`高`=最高价、`低`=最低价、`收`=最后一笔价、`量`=总成交量。

### 4. 日终清算

收盘后批处理，汇总全天：总成交量、总成交额、全天 OHLC。

## 四、关键决策

| 决策 | 原因 |
|------|------|
| 对账按 `tradeId` 关联 | 成交编号是双方共有的唯一键 |
| K 线用 `ts / bucketSeconds` 分桶 | 整数运算，桶起点可还原 |
| OHLC 增量更新 | 单次遍历 O(n)，high/low 取极值、close 取末值 |
| 金额整数运算 | 延续整数避免浮点原则 |

### `clearing/include/kline.hpp`

```cpp
#pragma once

#include "recon.hpp"

#include <algorithm>
#include <map>
#include <vector>

namespace clearing
{
    using Amount = int64_t;

    struct KLine
    {
        int64_t bucketStart = 0;
        Price open = 0, high = 0, low = 0, close = 0;
        Quantity volume = 0;
    };

    // 从成交聚合 K 线（按bucketSeconds分桶）
    inline std::vector<KLine> aggregateKLine(const std::vector<TradeRecord> &trades, int64_t bucketSeconds)
    {
        std::map<int64_t, KLine> buckets;
        for (const auto &t : trades)
        {
            int64_t start = (t.ts / bucketSeconds) * bucketSeconds;
            auto &k = buckets[start];
            if (k.volume == 0) // 新桶：open=high=low=第一笔价
            {
                k.bucketStart = start;
                k.open = k.high = k.low = t.price;
            }
            k.high = std::max(k.high, t.price);
            k.low = std::min(k.low, t.price);
            k.close = t.price;
            k.volume += t.qty;
        }
        std::vector<KLine> result;
        for (auto &kv : buckets)
            result.push_back(kv.second);
        return result;
    }

    // 日终清算汇总
    struct DailySummary
    {
        Quantity totalVolume = 0;
        Amount totalTurnover = 0;
        Price open = 0, high = 0, low = 0, close = 0;
    };

    inline DailySummary settle(const std::vector<TradeRecord> &trades, const std::string &symbol)
    {
        DailySummary s;
        bool first = true;
        for (const auto &t : trades)
        {
            if (t.symbol != symbol)
                continue;
            if (first)
            {
                s.open = s.high = s.low = t.price;
                first = false;
            }
            s.high = std::max(s.high, t.price);
            s.low = std::min(s.low, t.price);
            s.close = t.price;
            s.totalVolume += t.qty;
            s.totalTurnover += t.price * t.qty;
        }

        return s;
    }
} // namespace clearing```

### `clearing/include/recon.hpp`

```cpp
#pragma once

#include <cstdint>
#include <set>
#include <string>
#include <unordered_map>
#include <vector>

namespace clearing
{
    using Price = int64_t;
    using Quantity = int64_t;

    enum class Side
    {
        BUY,
        SELL
    };

    struct TradeRecord
    {
        std::string tradeId;
        std::string symbol;
        Side side;
        Price price;
        Quantity qty;
        int64_t ts;
    };

    // 内部对账结果
    struct PositionRecon
    {
        bool ok = false;
        Quantity recorded = 0;
        Quantity computed = 0;
    };

    // 外部对账差异
    struct TradeDiff
    {
        std::vector<std::string> missing;  // 交易所有、我方无
        std::vector<std::string> extra;    // 我方有、交易所无
        std::vector<std::string> mismatch; // 金额不符
    };

    // 内部对账：从成交明细重算持仓，与记录持仓对比
    inline PositionRecon reconcilePosition(const std::vector<TradeRecord> &trades, const std::string &symbol, Quantity recorded)
    {
        PositionRecon r;
        r.recorded = recorded;
        Quantity computed = 0;
        for (const auto &t : trades)
            if (t.symbol == symbol)
                computed += (t.side == Side::BUY) ? t.qty : -t.qty;
        r.computed = computed;
        r.ok = (computed == recorded);
        return r;
    }

    inline TradeDiff reconcileTrades(const std::vector<TradeRecord> &ours, const std::vector<TradeRecord> &exchange)
    {
        TradeDiff d;
        std::unordered_map<std::string, TradeRecord> oursMap;
        for (const auto &t : ours)
            oursMap[t.tradeId] = t;

        std::set<std::string> exchIds;
        for (const auto &t : exchange)
        {
            exchIds.insert(t.tradeId);
            auto it = oursMap.find(t.tradeId);
            if (it == oursMap.end())
                d.missing.push_back(t.tradeId);
            else if (it->second.price != t.price || it->second.qty != t.qty)
                d.mismatch.push_back(t.tradeId);
        }
        for (const auto &t : ours)
        {
            if (!exchIds.count(t.tradeId))
                d.extra.push_back(t.tradeId);
        }

        return d;
    }
} // namespace clearing```

### `clearing/src/main.cpp`

```cpp
#include "recon.hpp"
#include "kline.hpp"

#include <iostream>
#include <vector>

using namespace std;
using namespace clearing;

int main()
{
    // 成交明细（某标的）
    vector<TradeRecord> trades = {
        {"T1", "600000.SH", Side::BUY, 1000, 1000, 1},
        {"T2", "600000.SH", Side::SELL, 1000, 400, 2},
        {"T3", "600000.SH", Side::BUY, 1000, 500, 3},
    };

    // 内部对账：记录持仓 1100（正确）与 1000（错误）
    auto r1 = reconcilePosition(trades, "600000.SH", 1100);
    cout << "内部对账（记录1100）: " << (r1.ok ? "✅ 一致" : "❌ 不一致")
         << " 计算持仓=" << r1.computed << "\n";

    auto r2 = reconcilePosition(trades, "600000.SH", 1000);
    cout << "内部对账（记录1000）: " << (r2.ok ? "✅ 一致" : "❌ 不一致")
         << " 记录=" << r2.recorded << " 计算=" << r2.computed
         << " 差=" << (r2.computed - r2.recorded) << "\n";

    // 外部对账：我方 vs 交易所
    vector<TradeRecord> ours = {
        {"T1", "600000.SH", Side::BUY, 1000, 100, 1},
        {"T2", "600000.SH", Side::SELL, 1000, 50, 2},
        {"T3", "000001.SZ", Side::BUY, 2000, 100, 3},
    };
    vector<TradeRecord> exchange = {
        {"T1", "600000.SH", Side::BUY, 1000, 100, 1},
        {"T2", "600000.SH", Side::SELL, 1000, 60, 2}, // 数量不符
        {"T4", "600000.SH", Side::BUY, 1000, 30, 4},  // 交易所多出
    };

    auto d = reconcileTrades(ours, exchange);
    cout << "\n外部对账:\n";
    cout << "  缺失(交易所有/我方无): ";
    if (d.missing.empty())
        cout << "无";
    for (auto &id : d.missing)
        cout << id << " ";
    cout << "\n  多余(我方有/交易所无): ";
    if (d.extra.empty())
        cout << "无";
    for (auto &id : d.extra)
        cout << id << " ";
    cout << "\n  金额不符: ";
    if (d.mismatch.empty())
        cout << "无";
    for (auto &id : d.mismatch)
        cout << id << " ";
    cout << "\n";

    // ============ K线聚合 + 日终清算 ============
    std::cout << "\n===== K线聚合 + 日终清算 =====\n";

    std::vector<TradeRecord> day = {
        {"T1", "600000.SH", Side::BUY, 1000, 100, 60},
        {"T2", "600000.SH", Side::BUY, 1002, 50, 90},
        {"T3", "600000.SH", Side::SELL, 999, 80, 120},
        {"T4", "600000.SH", Side::BUY, 1001, 60, 150},
        {"T5", "600000.SH", Side::SELL, 1000, 30, 200},
    };

    // 1 分钟 K 线
    auto klines = aggregateKLine(day, 60);
    std::cout << "1 分钟 K 线:\n";
    for (const auto &k : klines)
        std::cout << "  " << k.bucketStart << "s: 开 " << k.open
                  << " 高 " << k.high << " 低 " << k.low
                  << " 收 " << k.close << " 量 " << k.volume << "\n";

    // 日终清算
    auto s = settle(day, "600000.SH");
    std::cout << "日终清算:\n"
              << "  开 " << s.open << " 高 " << s.high
              << " 低 " << s.low << " 收 " << s.close << "\n"
              << "  总成交量 " << s.totalVolume
              << " 总成交额 " << s.totalTurnover << "\n";

    return 0;
}```
