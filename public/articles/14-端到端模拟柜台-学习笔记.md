## 三、核心设计

### 1. 柜台门面（Facade 模式）

```
Counter（门面）
 ├─ RiskEngine（风控）      ← 下单前检查
 ├─ AccountManager（资金）  ← 冻结/扣款
 ├─ MatchingEngine（撮合）  ← 撮合成交
 └─ 持仓表                  ← 买卖更新
```

下单流程：**风控 → 冻结 → 撮合 → 双向结算 + 持仓更新**。

### 2. 模拟行情生成器

随机游走：每步在 ±N tick 内随机，价格围绕中间价波动。

### 3. 可观测性

- **指标**：下单数、拒单数、成交笔数、平均/最大延迟
- **健康检查**：一行汇总当前状态

## 四、关键决策

| 决策 | 原因 |
|------|------|
| 柜台用门面模式整合各模块 | 对外一个简洁 API，对内屏蔽复杂度 |
| 指标在调用点记录延迟 | 精准测量单笔处理耗时 |
| 模块源码直接编译复用 | 避免跨项目链接依赖 |

### `counter/include/market_data.hpp`

```cpp
#pragma once

#include <cstdint>
#include <random>

namespace counter
{
    // 随机游走行情生成器
    class MarketDataGenerator
    {
    public:
        explicit MarketDataGenerator(int64_t mid, uint32_t seed = 42) : mid_(mid), rng_(seed) {}
        int64_t next()
        {
            std::uniform_int_distribution<int> d(-3, 3); // ±3 tick
            mid_ += d(rng_);
            if (mid_ < 1)
            {
                mid_ = 1;
            }
            return mid_;
        }

        int64_t mid() const { return mid_; }

    private:
        int64_t mid_;
        std::mt19937 rng_;
    };
} // namespace counter```

### `counter/include/metrics.hpp`

```cpp
#pragma once

#include <cstdint>
#include <string>

namespace counter
{

    // 可观测性指标
    struct Metrics
    {
        uint64_t orders = 0;         // 下单数
        uint64_t rejected = 0;       // 拒单数
        uint64_t trades = 0;         // 成交笔数
        uint64_t totalLatencyNs = 0; // 总延迟
        uint64_t maxLatencyNs = 0;

        void record(bool rej, uint64_t latNs, int tradeCount)
        {
            ++orders;
            if (rej)
                ++rejected;
            trades += static_cast<uint64_t>(tradeCount);
            totalLatencyNs += latNs;
            if (latNs > maxLatencyNs)
                maxLatencyNs = latNs;
        }

        double avgLatencyUs() const
        {
            return orders ? static_cast<double>(totalLatencyNs) / orders / 1000.0 : 0.0;
        }

        // 健康检查
        std::string health() const
        {
            return "orders=" + std::to_string(orders) +
                   " rejected=" + std::to_string(rejected) +
                   " trades=" + std::to_string(trades) +
                   " avgLat=" + std::to_string(avgLatencyUs()) + "us" +
                   " maxLat=" + std::to_string(static_cast<double>(maxLatencyNs) / 1000.0) + "us";
        }
    };

} // namespace counter```

### `counter/src/counter.cpp`

```cpp
#include "counter.hpp"

namespace counter
{

    Counter::Counter() : engine_(4)
    {
        // 默认预交易规则
        riskEngine_.addRule(std::make_shared<risk::FatFingerRule>(1000000));
        riskEngine_.addRule(std::make_shared<risk::PriceRangeRule>(1, 100000000));
        riskEngine_.addRule(std::make_shared<risk::PositionLimitRule>(1000000));
        riskEngine_.addRule(std::make_shared<risk::FundLimitRule>());
    }

    bool Counter::deposit(const std::string &userId, Amount amount)
    {
        return accounts_.deposit(userId, amount);
    }

    me::MatchingEngine::OrderResult Counter::placeOrder(const std::string &userId,
                                                        const std::string &symbol,
                                                        Side side, Price price, Quantity qty)
    {
        me::MatchingEngine::OrderResult result;

        me::Order o;
        o.id = nextOrderId_++;
        o.symbol = symbol;
        o.side = side;
        o.type = me::OrderType::LIMIT;
        o.price = price;
        o.totalQty = qty;

        // 1. 预交易风控
        risk::RiskContext ctx;
        ctx.userId = userId;
        ctx.position = position(userId, symbol);
        ctx.availableFunds = accounts_.account(userId).availableAmount;
        std::string reason;
        const char *rule = riskEngine_.checkPreTrade(o, ctx, reason);
        if (rule)
        {
            result.rejected = true;
            result.rejectReason = std::string(rule) + ": " + reason;
            result.order = o;
            return result;
        }

        // 2. 买入冻结资金
        if (side == me::Side::BUY)
        {
            if (!accounts_.freeze(userId, price * qty))
            {
                result.rejected = true;
                result.rejectReason = "可用资金不足";
                result.order = o;
                return result;
            }
        }

        // 3. 记录订单
        OrderInfo info;
        info.userId = userId;
        info.symbol = symbol;
        info.side = side;
        if (side == me::Side::BUY)
            info.frozenAmount = price * qty;
        orders_[o.id] = info;

        // 4. 撮合
        result = engine_.process(o);

        // 5. 成交双向结算 + 持仓更新
        for (const auto &t : result.trades)
        {
            if (side == me::Side::BUY) // taker 买入
            {
                accounts_.deduct(userId, t.price * t.qty);
                orders_[o.id].deductedAmount += t.price * t.qty;
                positions_[userId][symbol] += t.qty;
            }
            else // taker 卖出
            {
                accounts_.credit(userId, t.price * t.qty);
                positions_[userId][symbol] -= t.qty;
            }

            auto it = orders_.find(t.makerId); // maker 是对方
            if (it != orders_.end())
            {
                if (it->second.side == me::Side::BUY)
                {
                    accounts_.deduct(it->second.userId, t.price * t.qty);
                    it->second.deductedAmount += t.price * t.qty;
                    positions_[it->second.userId][symbol] += t.qty;
                }
                else
                {
                    accounts_.credit(it->second.userId, t.price * t.qty);
                    positions_[it->second.userId][symbol] -= t.qty;
                }
            }
        }

        return result;
    }

    bool Counter::cancel(const std::string &userId, OrderId id)
    {
        auto it = orders_.find(id);
        if (it == orders_.end())
            return false;
        bool ok = engine_.cancel(it->second.symbol, id);
        if (ok && it->second.side == me::Side::BUY)
        {
            Amount remaining = it->second.frozenAmount - it->second.deductedAmount;
            accounts_.unfreeze(userId, remaining);
        }
        return ok;
    }

    Amount Counter::availableFunds(const std::string &userId) const
    {
        return accounts_.account(userId).availableAmount;
    }

    Quantity Counter::position(const std::string &userId, const std::string &symbol) const
    {
        auto u = positions_.find(userId);
        if (u == positions_.end())
            return 0;
        auto s = u->second.find(symbol);
        return s != u->second.end() ? s->second : 0;
    }

    me::SymbolBook::Snapshot Counter::snapshot(const std::string &symbol) const
    {
        return engine_.snapshot(symbol);
    }

} // namespace counter```

### `counter/src/main.cpp`

```cpp
#include "counter.hpp"
#include "market_data.hpp"
#include "metrics.hpp"

#include <chrono>
#include <iostream>

using namespace std;
using namespace counter;
using namespace me;

static uint64_t getCurrentTimeInNanoseconds()
{
     return chrono::duration_cast<chrono::nanoseconds>(
                chrono::steady_clock::now().time_since_epoch())
         .count();
}

int main()
{
     cout << "============================================\n";
     cout << "  模拟柜台（仿真柜台）端到端演示\n";
     cout << "============================================\n\n";

     Counter c;
     Metrics m;
     MarketDataGenerator gen(1000);

     // 1. 开户 + 入金
     cout << "【1】开户 + 入金\n";
     c.deposit("alice", 100000);
     c.deposit("bob", 100000);
     cout << "  alice 可用 " << c.availableFunds("alice")
          << "，bob 可用 " << c.availableFunds("bob") << "\n\n";

     // 2. 模拟行情
     cout << "【2】模拟行情（随机游走）\n";
     for (int i = 0; i < 3; ++i)
          gen.next();
     cout << "  最新价: " << gen.mid() << "\n\n";

     // 3. 下单（风控 → 冻结 → 撮合）
     cout << "【3】下单\n";
     uint64_t t0 = getCurrentTimeInNanoseconds();
     auto r1 = c.placeOrder("alice", "600000.SH", Side::BUY, 1000, 100);
     uint64_t t1 = getCurrentTimeInNanoseconds();
     m.record(r1.rejected, t1 - t0, r1.trades.size());
     cout << "  alice 买 100@1000 → " << (r1.rejected ? "拒绝" : "挂单")
          << "（延迟 " << (t1 - t0) / 1000 << " us）\n";

     t0 = getCurrentTimeInNanoseconds();
     auto r2 = c.placeOrder("bob", "600000.SH", Side::SELL, 1000, 40);
     t1 = getCurrentTimeInNanoseconds();
     m.record(r2.rejected, t1 - t0, r2.trades.size());
     cout << "  bob 卖 40@1000 → 成交 " << r2.order.filledQty << " 股\n";

     t0 = getCurrentTimeInNanoseconds();
     auto r3 = c.placeOrder("alice", "600000.SH", Side::BUY, 1000, 1000);
     t1 = getCurrentTimeInNanoseconds();
     m.record(r3.rejected, t1 - t0, r3.trades.size());
     cout << "  alice 买 1000@1000 → "
          << (r3.rejected ? "拒绝(" + r3.rejectReason + ")" : "通过") << "\n\n";

     // 4. 查询（柜台对外接口）
     cout << "【4】查询（账户/持仓/盘口）\n";
     cout << "  alice: 可用 " << c.availableFunds("alice")
          << " 持仓 " << c.position("alice", "600000.SH") << "\n";
     cout << "  bob:   可用 " << c.availableFunds("bob")
          << " 持仓 " << c.position("bob", "600000.SH") << "\n";
     auto snap = c.snapshot("600000.SH");
     cout << "  盘口 600000.SH:";
     if (snap.bids.empty())
          cout << " 买(空)";
     for (auto &lv : snap.bids)
          cout << " 买" << lv.price << "x" << lv.qty;
     if (snap.asks.empty())
          cout << " 卖(空)";
     for (auto &lv : snap.asks)
          cout << " 卖" << lv.price << "x" << lv.qty;
     cout << "\n\n";

     // 5. 可观测性
     cout << "【5】可观测性（健康检查）\n";
     cout << "  " << m.health() << "\n\n";

     cout << "============================================\n";
     cout << "  链路走通：开户→入金→下单→成交→持仓→查询→指标\n";
     cout << "============================================\n";
     return 0;
}```
